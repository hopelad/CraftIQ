 489    function ciHandleFile(file) {
 490        if (!file) return;
 491        var ext = (file.name.split(".").pop() || "").toLowerCase();
 492        var isImage = ["png","jpg","jpeg","webp"].includes(ext);
 493        var isDoc   = ["pdf","docx","txt"].includes(ext);
 494        if (!isImage && !isDoc) {
 495            alert("Unsupported file type. Please use PDF, DOCX, TXT, PNG, JPG, JPEG, or WEBP.");
 496            return;
 497        }
 498
 499        ciShowFileInfo(file.name, isImage ? "&#128247;" : "&#128196;", "Extracting…");
 500        document.getElementById("ciStep1Btn").disabled = true;
 501        _ciReady = false;
 502        _cicUploadedText = null;
 503        _cicUploadCache  = null;
 504
 505        if (ext === "txt") {
 506            var reader = new FileReader();
 507            reader.onload = function(e) {
 508                _cicUploadedText = (e.target.result || "").trim();
 509                _ciReady = true;
 510                ciShowFileInfo(file.name, "&#128196;", "✓ Ready");
 511                document.getElementById("ciStep1Btn").disabled = false;
 512                document.getElementById("ciOcrBox").style.display = "none";
 513            };
 514            reader.readAsText(file);
 515        } else {
 516            var fd = new FormData();
 517            fd.append("file", file);
 518            fetch("/api/cv/extract", { method: "POST", body: fd })
 519            .then(function(r) { return r.json().then(function(d) { return { ok: r.ok, data: d }; }); })
 520            .then(function(r) {
 521                if (!r.ok) throw new Error(r.data.message || "Extraction failed.");
 522                _cicUploadedFormData = r.data.formData;
 523                var extracted = cicFormDataToText(r.data.formData);
 524
 525                if (isImage) {
 526                    // Show OCR review before confirming
 527                    document.getElementById("ciOcrText").value = extracted;
 528                    document.getElementById("ciOcrBox").style.display = "block";
 529                    ciShowFileInfo(file.name, "&#128247;", "Review text below");
 530                    // User must click Apply before continuing
 531                } else {

 538            })

 541            });
